## Acceptance Criteria

### AC1: Race Condition Fix (P0 - BACK-001)
1. Implement transaction with SELECT FOR UPDATE or optimistic locking
2. Add unique constraint validation at database level
3. Test with concurrent booking requests (no double bookings)
4. Proper error handling for capacity exceeded

### AC2: Auth Error Handling (P0 - BACK-002)
1. Wrap error throw in proper try/catch
2. Return ActionResult error instead of throwing
3. Graceful error messages for users

### AC3: TimeSlot Validation (P1 - BACK-003)
1. Validate "HH:mm" format before parsing
2. Throw explicit error for invalid format
3. Add Zod schema for timeSlot format
4. No more NaN from parseInt

### AC4: Email Retry Queue (P1 - BACK-004)
1. Implement retry logic for failed emails (3 attempts, exponential backoff)
2. Log failed emails for admin review
3. Non-blocking email sending

### AC5: Sentry Integration (P2 - BACK-005)
1. Add `Sentry.captureException()` to all catch blocks
2. Configure Sentry DSN in production
3. Add context (userId, bookingId) to errors
4. Error tracking dashboard accessible

### AC6: Email URL Fix (P2 - BACK-006)
1. Replace hardcoded `'https://encave.ch/'` with `getBaseUrl()`
2. Support preview/staging environments
3. Test email links in different environments

### AC7: Structured Logging (P2 - BACK-007)
1. Implement structured logger (Pino or Winston)
2. Add log levels (error, warn, info, debug)
3. Add context fields (userId, action, timestamp)
4. Replace console.log/error with logger calls

// Race condition fix with transaction
async function createBookingWithLock(input: BookingInput) {
  return db.$transaction(async (tx) => {
    // Lock the slots for this experience/date/time
    const existingBookings = await tx.booking.findMany({
      where: {
        experienceId: input.experienceId,
        date: input.date,
        timeSlot: input.timeSlot,
        status: { in: ['PENDING_PAYMENT', 'CONFIRMED'] }
      },
      select: { guestCount: true }
    });

    const bookedCount = existingBookings.reduce((sum, b) => sum + b.guestCount, 0);
    const remainingCapacity = experience.maxCapacity - bookedCount;

    if (input.guestCount > remainingCapacity) {
      throw new Error('NO_CAPACITY');
    }

    // Create booking within same transaction
    return tx.booking.create({ data: { ... } });
  }, {
    isolationLevel: 'Serializable' // Prevents race conditions
  });
}

// Email retry with exponential backoff
async function sendEmailWithRetry(
  sendFn: () => Promise<void>,
  maxAttempts = 3
) {
  for (let attempt = 1; attempt <= maxAttempts; attempt++) {
    try {
      await sendFn();
      return; // Success
    } catch (error) {
      if (attempt === maxAttempts) {
        logger.error({ error, attempt }, 'Email send failed after max retries');
        throw error;
      }
      const delay = Math.pow(2, attempt - 1) * 1000; // 1s, 2s, 4s
      await new Promise(resolve => setTimeout(resolve, delay));
    }
  }
}

export function getBaseUrl(): string {
  if (process.env.VERCEL_URL) {
    return `https://${process.env.VERCEL_URL}`;
  }
  if (process.env.NEXT_PUBLIC_APP_URL) {
    return process.env.NEXT_PUBLIC_APP_URL;
  }
  return 'http://localhost:3000';
}
