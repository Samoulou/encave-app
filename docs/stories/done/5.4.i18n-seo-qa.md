### hreflang Implementation

```typescript
// src/app/[locale]/layout.tsx
import { Metadata } from 'next';

export async function generateMetadata({
  params: { locale },
}: {
  params: { locale: string };
}): Promise<Metadata> {
  const baseUrl = 'https://encave.ch';

  return {
    alternates: {
      canonical: `${baseUrl}/${locale}`,
      languages: {
        'fr': `${baseUrl}/fr`,
        'de': `${baseUrl}/de`,
        'en': `${baseUrl}/en`,
        'x-default': `${baseUrl}/fr`,
      },
    },
  };
}
```

### Localized Metadata Example

```typescript
// src/app/[locale]/experiences/page.tsx
import { getTranslations } from 'next-intl/server';

export async function generateMetadata({
  params: { locale },
}: {
  params: { locale: string };
}) {
  const t = await getTranslations({ locale, namespace: 'meta' });

  return {
    title: t('experiences.title'),
    description: t('experiences.description'),
    openGraph: {
      locale: locale === 'fr' ? 'fr_CH' : locale === 'de' ? 'de_CH' : 'en_US',
    },
  };
}
```

### Sitemap with hreflang

```xml
<!-- public/sitemap.xml -->
<?xml version="1.0" encoding="UTF-8"?>
<urlset xmlns="http://www.sitemaps.org/schemas/sitemap/0.9"
        xmlns:xhtml="http://www.w3.org/1999/xhtml">
  <url>
    <loc>https://encave.ch/fr/experiences</loc>
    <xhtml:link rel="alternate" hreflang="fr" href="https://encave.ch/fr/experiences"/>
    <xhtml:link rel="alternate" hreflang="de" href="https://encave.ch/de/experiences"/>
    <xhtml:link rel="alternate" hreflang="en" href="https://encave.ch/en/experiences"/>
    <xhtml:link rel="alternate" hreflang="x-default" href="https://encave.ch/fr/experiences"/>
  </url>
</urlset>
```

### Translation Coverage Script

```typescript
// scripts/check-translations.ts
import fr from '../messages/fr.json';
import de from '../messages/de.json';
import en from '../messages/en.json';

function getKeys(obj: object, prefix = ''): string[] {
  return Object.entries(obj).flatMap(([key, value]) => {
    const newKey = prefix ? `${prefix}.${key}` : key;
    if (typeof value === 'object' && value !== null) {
      return getKeys(value, newKey);
    }
    return [newKey];
  });
}

const frKeys = new Set(getKeys(fr));
const deKeys = new Set(getKeys(de));
const enKeys = new Set(getKeys(en));

const missingInDe = [...frKeys].filter(k => !deKeys.has(k));
const missingInEn = [...frKeys].filter(k => !enKeys.has(k));

if (missingInDe.length || missingInEn.length) {
  console.error('Missing translations:');
  console.error('DE:', missingInDe);
  console.error('EN:', missingInEn);
  process.exit(1);
}

console.log('All translations complete!');
```
