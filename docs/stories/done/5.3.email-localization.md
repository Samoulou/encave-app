### Email Service Update

```typescript
// src/lib/email/send-email.ts
import { Resend } from 'resend';

const resend = new Resend(process.env.RESEND_API_KEY);

export async function sendBookingConfirmation(
  to: string,
  locale: 'fr' | 'de' | 'en',
  data: BookingData
) {
  const Template = await import(`@/emails/templates/booking-confirmation/${locale}`);

  const subject = {
    fr: 'Confirmation de votre reservation',
    de: 'Bestatigung Ihrer Reservierung',
    en: 'Your booking confirmation',
  }[locale];

  await resend.emails.send({
    from: 'EnCave <noreply@encave.ch>',
    to,
    subject,
    react: Template.default(data),
  });
}
```

### Sample Email Template

```tsx
// src/emails/templates/booking-confirmation/fr.tsx
import { Html, Head, Body, Container, Text } from '@react-email/components';

interface BookingConfirmationProps {
  guestName: string;
  experienceTitle: string;
  date: Date;
  time: string;
  guestCount: number;
}

export default function BookingConfirmationFR({
  guestName,
  experienceTitle,
  date,
  time,
  guestCount,
}: BookingConfirmationProps) {
  const formattedDate = new Intl.DateTimeFormat('fr-CH', {
    weekday: 'long',
    day: 'numeric',
    month: 'long',
    year: 'numeric',
  }).format(date);

  return (
    <Html>
      <Head />
      <Body>
        <Container>
          <Text>Bonjour {guestName},</Text>
          <Text>
            Votre reservation pour "{experienceTitle}" est confirmee.
          </Text>
          <Text>
            Date: {formattedDate} a {time}
          </Text>
          <Text>
            Nombre de personnes: {guestCount}
          </Text>
          <Text>A bientot!</Text>
          <Text>L'equipe EnCave</Text>
        </Container>
      </Body>
    </Html>
  );
}
```
